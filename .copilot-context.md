## 🎯 Project Context

**Zoi** is a **Docker Compose-based AI ecosystem** with **microservices architecture**. This file helps code completion tools understand our patterns and suggest appropriate code.

## 🏗️ Code Patterns & Templates

### Docker Compose Service Template
```yaml
# Standard service definition pattern
service-name:
  image: official/image:latest
  container_name: service-name
  restart: unless-stopped
  ports:
    - "${SERVICE_PORT:-8080}:8080"
  environment:
    - NODE_ENV=production
    - DATABASE_URL=postgresql://postgres:password@postgres:5432/dbname
    - REDIS_URL=redis://redis:6379
    - API_KEY=${API_KEY}
  volumes:
    - service-data:/app/data
    - ../../config/service:/config:ro
    - /etc/localtime:/etc/localtime:ro
  networks:
    - zoi-network
  depends_on:
    - postgres
    - redis
  healthcheck:
    test: ["CMD-SHELL", "curl -f http://localhost:8080/health || exit 1"]
    interval: 30s
    timeout: 10s
    retries: 3
    start_period: 40s
  labels:
    - "traefik.enable=true"
    - "traefik.http.routers.service.rule=Host(`service.${DOMAIN:-zoi.local}`)"
    - "traefik.http.services.service.loadbalancer.server.port=8080"
```

### FastAPI Service Template
```python
from fastapi import FastAPI, HTTPException, Depends
from pydantic import BaseModel
import os
import logging

# Configuration
app = FastAPI(title="Service Name", version="1.0.0")
logger = logging.getLogger(__name__)

# Environment variables
DATABASE_URL = os.getenv("DATABASE_URL")
REDIS_URL = os.getenv("REDIS_URL")
LITELLM_API_KEY = os.getenv("LITELLM_API_KEY")

# Pydantic models
class RequestModel(BaseModel):
    prompt: str
    model: str = "zoi-thinker"
    max_tokens: int = 150

class ResponseModel(BaseModel):
    response: str
    model_used: str
    tokens_used: int

# Health check endpoint
@app.get("/health")
async def health_check():
    return {"status": "healthy", "service": "service-name"}

# Main endpoint
@app.post("/api/endpoint", response_model=ResponseModel)
async def process_request(request: RequestModel):
    try:
        # Process request
        result = await process_with_litellm(request)
        return ResponseModel(**result)
    except Exception as e:
        logger.error(f"Error processing request: {e}")
        raise HTTPException(status_code=500, detail=str(e))

# LiteLLM integration
async def process_with_litellm(request: RequestModel):
    import httpx
    
    async with httpx.AsyncClient() as client:
        response = await client.post(
            "http://litellm:4000/v1/chat/completions",
            headers={"Authorization": f"Bearer {LITELLM_API_KEY}"},
            json={
                "model": request.model,
                "messages": [{"role": "user", "content": request.prompt}],
                "max_tokens": request.max_tokens
            }
        )
        return response.json()
```

### Qdrant Vector Operations
```python
import httpx

# Search vectors
async def search_vectors(query_vector: list, collection: str, limit: int = 5):
    async with httpx.AsyncClient() as client:
        response = await client.post(

                "vector": query_vector,
                "limit": limit,
                "with_payload": True,

        response = await client.put(
            f"http://qdrant:6333/collections/{collection}/points",
            json={"points": points}
        )
        return response.json()
```

## 🚨 Common Error Patterns & Solutions

### Service Connection Errors
```python
# Retry pattern for service connections
import asyncio
from functools import wraps

def retry_on_connection_error(max_retries: int = 3, delay: float = 1.0):
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            for attempt in range(max_retries):
                try:
                    return await func(*args, **kwargs)
                except (httpx.ConnectError, httpx.TimeoutException) as e:
                    if attempt == max_retries - 1:
                        raise
                    await asyncio.sleep(delay * (2 ** attempt))
            return None
        return wrapper
    return decorator

@retry_on_connection_error(max_retries=3)
async def call_external_service():
    # Service call implementation
    pass
```

### Configuration Validation
```python
import os
from typing import Optional

def validate_environment():
    """Validate required environment variables"""
    required_vars = [
        "DATABASE_URL",
        "REDIS_URL", 
        "LITELLM_API_KEY"
    ]
    
    missing_vars = []
    for var in required_vars:
        if not os.getenv(var):
            missing_vars.append(var)
    
    if missing_vars:
        raise ValueError(f"Missing required environment variables: {missing_vars}")

# Call during startup
validate_environment()
```
